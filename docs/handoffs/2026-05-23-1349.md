## 3. Qué sigue — Plan detallado de Sistema de Momentum

### Orden de ejecución sugerido

**Task 1: `MomentumSystem` autoload (lógica core).**
Archivo: `scripts/systems/momentum_system.gd`.

Estructura:
```gdscript
extends Node
# Autoload "MomentumSystem"

const MAX_LEVEL := 10
const BLOCK_FREEZE_SECONDS := 1.0

signal momentum_changed(new_level: int)
signal momentum_reset()
signal threshold_5x_reached()
signal threshold_5x_lost()

var current_level: int = 0
var _frozen: bool = false
var _freeze_timer: float = 0.0

func _process(delta: float) -> void:
    if _frozen:
        _freeze_timer -= delta
        if _freeze_timer <= 0.0:
            _frozen = false

func on_hit_landed() -> void:
    if current_level >= MAX_LEVEL:
        return
    var was_below_5: bool = current_level < 5
    current_level += 1
    momentum_changed.emit(current_level)
    if was_below_5 and current_level >= 5:
        threshold_5x_reached.emit()

func on_damage_taken() -> void:
    if _frozen:
        return  # bloqueo congela, no resetea
    if current_level == 0:
        return
    var was_at_5_plus: bool = current_level >= 5
    current_level = 0
    momentum_changed.emit(0)
    momentum_reset.emit()
    if was_at_5_plus:
        threshold_5x_lost.emit()

func on_block_absorbed() -> void:
    # Bloqueo congela Momentum 1s (no lo resetea).
    _frozen = true
    _freeze_timer = BLOCK_FREEZE_SECONDS

func damage_multiplier() -> float:
    return 1.0 + 0.05 * float(current_level)

func drop_multiplier() -> float:
    return 1.0 + 0.1 * float(current_level)

func furia_multiplier() -> float:
    return 1.0 + 0.05 * float(current_level)
```
